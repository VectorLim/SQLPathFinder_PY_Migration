SELECT  DISTINCT 
          ww AS ww
         ,prod AS prod
         ,Replace(Replace(Replace(Replace(Replace(Replace(subs,',',';'),chr(9),' '),chr(10),' '),chr(13),' '),chr(34),''''),chr(7),' ') AS subs
         ,lot AS lot
         ,qty AS qty
         ,To_Char(out_date,'yyyy-mm-dd hh24:mi:ss') AS out_date
         ,attribute_name AS attribute_name
         ,owner AS owner
         ,To_Char(SYSDATE,'yyyy-mm-dd hh24:mi:ss') AS current_time
FROM
(
SELECT  
          c0.ww AS ww
         ,p.prodgroup3 AS prod
         ,f12.attribute_value AS subs
         ,f0.lot AS lot
         ,f0.newqty1 AS qty
         ,f0.out_date AS out_date
         ,(SELECT alpha1 as attribute_name FROM @[]@F_GTSData WHERE facility IN ('IMORA','IMO','COMETS') AND table_name = 'COM$ATTR' AND key1 = 'LOT' AND src_erase_date IS NULL AND key2_numeric = f12.attribute_number AND rownum <= 1) AS attribute_name
         ,f0.owner AS owner
FROM 
@[]@F_LotHist f0
INNER JOIN @[]@F_Calendar c0 ON f0.last_action_date BETWEEN c0.start_date AND c0.end_date AND c0.event_code = 'S' AND decode(f0.facility,'RA3','AAL',f0.facility)= c0.facility
LEFT JOIN @[]@F_Product p ON p.product = f0.product AND p.facility = f0.facility AND NVL(p.latest_version,'Y') = 'Y' -- AND p.product_version = f0.product_version
INNER JOIN @[]@F_Lot f9 ON f9.lot = f0.lot
LEFT JOIN @[]@F_LotAttribute f12 ON f9.lot = f12.lot
WHERE
NVL(f0.history_deleted_flag,'N') = 'N'
AND      f0.owner <> 'EMPTYFOUP'
 AND      f0.operation = '1210' 
 AND      f0.last_pass = 'Y' 
 AND      f0.movedout = 'Y' 
 AND      f0.owner = 'PROD' 
 AND      f0.out_date Between To_Date('28-Jun-2026 00:00:00','dd-Mon-yyyy hh24:mi:ss') AND To_Date('30-Jun-2026 00:00:00','dd-Mon-yyyy hh24:mi:ss') 
-- Tail A
)
WHERE
              attribute_name = 'SUB VEND ID' 
/*END SQL*/

SELECT /*L4*/  DISTINCT 
          [ww] AS [ww]
         ,[link] AS [link]
         ,[prod] AS [prod]
         ,Replace(Replace(Replace(Replace(Replace(Replace([subs],',',';'),CAST(X'09' AS TEXT),' '),CAST(X'0A' AS TEXT),' '),CAST(X'0D' AS TEXT),' '),CAST(X'22' AS TEXT),''''),CAST(X'07' AS TEXT),' ') AS [subs]
         ,[lot] AS [lot]
         ,[qty] AS [qty]
         ,[out_date] AS [out_date]
         ,[current_time] AS [current_time]
         ,[LOT_GROUP] AS [LOT_GROUP]
         ,CASE WHEN 2=0 THEN CAST(ROUND([time_gap],0) AS Integer) ELSE Round([time_gap],2) END AS [time_gap]
         ,[SEQ] AS [SEQ]
         ,[lot_2] AS [lot_2]
FROM
(
SELECT /*L3*/ 
          [ww] AS [ww]
         ,[link] AS [link]
         ,[prod] AS [prod]
         ,[subs] AS [subs]
         ,[lot] AS [lot]
         ,[qty] AS [qty]
         ,[out_date] AS [out_date]
         ,[attribute_name] AS [attribute_name]
         ,[owner] AS [owner]
         ,[lot_1] AS [lot_1]
         ,[program_name] AS [program_name]
         ,[current_time] AS [current_time]
         ,[LOT_GROUP] AS [LOT_GROUP]
         ,[time_gap] AS [time_gap]
         ,RANK () OVER (PARTITION BY  [LOT_GROUP]  ORDER BY  [out_date]  DESC) AS [SEQ]
         ,MIN ( [time_gap] ) OVER (PARTITION BY  [LOT_GROUP]  ) AS [MIN_TIME]
         ,[lot_2] AS [lot_2]
FROM
(
SELECT /*L2*/ 
          [ww] AS [ww]
         ,[link] AS [link]
         ,[prod] AS [prod]
         ,[subs] AS [subs]
         ,[lot] AS [lot]
         ,[qty] AS [qty]

         ,[attribute_name] AS [attribute_name]
         ,[owner] AS [owner]
         ,[lot_1] AS [lot_1]
         ,[program_name] AS [program_name]
         ,[current_time] AS [current_time]
         ,[link] ||'_'|| [prod] ||'_'|| [subs] AS [LOT_GROUP]
         ,[time_gap] AS [time_gap]
         ,[lot_2] AS [lot_2]
FROM
(
SELECT /*L1*/ 
          [ww] AS [ww]
         ,[link] AS [link]
         ,[prod] AS [prod]
         ,[subs] AS [subs]
         ,[lot] AS [lot]
         ,[qty] AS [qty]
         ,[out_date] AS [out_date]
         ,[attribute_name] AS [attribute_name]
         ,[owner] AS [owner]
         ,[lot_1] AS [lot_1]
         ,[program_name] AS [program_name]
         ,[current_time] AS [current_time]
         ,[time_gap] AS [time_gap]
         ,[lot_2] AS [lot_2]
FROM
(
SELECT /*L0*/  
          a0.[ww] AS [ww]
         ,a1.[link] AS [link]
         ,a0.[prod] AS [prod]
         ,a0.[subs] AS [subs]
         ,a0.[lot] AS [lot]
         ,a0.[qty] AS [qty]
         ,a0.[out_date] AS [out_date]
         ,a0.[attribute_name] AS [attribute_name]
         ,a0.[owner] AS [owner]
         ,a1.[lot_1] AS [lot_1]
         ,a1.[program_name] AS [program_name]
         ,a0.[current_time] AS [current_time]
         ,(( strftime('%s',[current_time]) -  strftime('%s',[out_date]) ) / 3600.0 / 24.0)*24 AS [time_gap]
